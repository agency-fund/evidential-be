17def _icc_one_way_random_intercept(df: pd.DataFrame, *, cluster_column: str, outcome_column: str) -> float: 

18 """ 

19 ICC for a single-intercept random-cluster model (outcome ~ 1 + (1|cluster)). 

20 

21 Estimated from one-way random-effects ANOVA mean squares (Method of Moments): 

22 

23 ICC = σ²_between / (σ²_between + σ²_within) = (MSB - MSW) / (MSB + (n0 - 1) * MSW) 

24 

25 where n0 is the adjustment for unequal cluster sizes. ICC estimates are clamped to [0, 1]. 

26 """ 

27 # 1. cluster stats 

28 n_total = len(df) 

29 y = df[outcome_column].to_numpy(dtype=np.float64) 

30 codes, _ = pd.factorize(df[cluster_column]) # indices of cluster membership 

31 cluster_sizes = np.bincount(codes).astype(np.float64) 

32 k = len(cluster_sizes) 

33 

34 # 2. ANOVA components for ICC 

35 df_b = k - 1 

36 df_w = n_total - k 

37 if df_w < 1: 

38 raise ValueError("Insufficient within-cluster data (need N > clusters)") 

39 

40 cluster_sums = np.bincount(codes, weights=y) 

41 cluster_means = cluster_sums / cluster_sizes 

42 grand_mean = np.mean(y) 

43 # Sum of Squares Between (SSB) 

44 ssb = np.sum(cluster_sizes * (cluster_means - grand_mean) ** 2) 

45 # Sum of Squares Within (SSW) - subtract the mean of each cluster from each observation 

46 ssw = np.sum((y - cluster_means[codes]) ** 2) 

47 # Derive the mean squares 

48 msb = ssb / df_b 

49 msw = ssw / df_w 

50 

51 # 3. n0 calculation; can think of it as an "effective cluster size" when we have unequal sizes 

52 # for an unbiased estimate of ICC. n0 equals the common cluster size when sizes are balanced. 

53 sum_n2 = np.sum(cluster_sizes**2) 

54 n0 = (n_total - (sum_n2 / n_total)) / df_b 

55 

56 # 4. Final calculation of ICC as the ratio of between-cluster variance to total variance. 

57 # If MSB < MSW, the point estimate for sigma_b is 0 (prevents negative ICC) 

58 var_between = max(0.0, (msb - msw) / n0) 

59 var_within = msw 

60 total_var = var_between + var_within 

61 if total_var == 0: 

62 return 0.0 

63 

64 return float(np.clip(var_between / total_var, 0.0, 1.0))

67def calculate_icc_from_sufficient_stats( 

68 *, 

69 counts: _FloatArray, 

70 sums: _FloatArray, 

71 sumsqs: _FloatArray, 

72) -> float: 

73 """ 

74 Calculate intraclass correlation (ICC) from per-cluster sufficient statistics. 

75 

76 Computes the same one-way random-effects ANOVA estimator as 

77 calculate_icc_from_dataframe, but from one (count, sum, sum-of-squares) triple per 

78 cluster instead of individual observations. The sums may be over values shifted by 

79 any per-metric constant (ICC is shift-invariant); all three arrays must describe the 

80 same non-null observations and use the same shift. 

81 

82 Args: 

83 counts: Number of non-null observations in each cluster; all must be > 0 

84 sums: Sum of the observations in each cluster 

85 sumsqs: Sum of the squared observations in each cluster 

86 

87 Returns: 

88 ICC value between 0 and 1 

89 

90 Raises: 

91 ValueError: If there are fewer than 2 clusters or insufficient observations 

92 """ 

93 n = np.asarray(counts, dtype=np.float64) 

94 s = np.asarray(sums, dtype=np.float64) 

95 q = np.asarray(sumsqs, dtype=np.float64) 

96 if not (len(n) == len(s) == len(q)): 

97 raise ValueError("counts, sums, and sumsqs must have the same length") 

98 if np.any(n <= 0): 

99 raise ValueError("counts must all be positive; drop clusters without observations") 

100 

101 k = len(n) 

102 if k < 2: 

103 raise ValueError("Need at least 2 clusters to calculate ICC") 

104 n_total = float(n.sum()) 

105 df_b = k - 1 

106 df_w = n_total - k 

107 if df_w < 1: 

108 raise ValueError("Insufficient within-cluster data (need N > clusters)") 

109 

110 # ANOVA sums of squares via the identity sum((x - mean)^2) = sum(x^2) - sum(x)^2 / n. 

111 # Each term is a sum of squares, so clamp tiny negative floating-point results to 0. 

112 between_terms = s**2 / n 

113 ssw = max(0.0, float(q.sum() - between_terms.sum())) 

114 ssb = max(0.0, float(between_terms.sum() - s.sum() ** 2 / n_total)) 

115 msb = ssb / df_b 

116 msw = ssw / df_w 

117 

118 # n0 is the "effective cluster size" adjustment for unequal cluster sizes. 

119 n0 = (n_total - float((n**2).sum()) / n_total) / df_b 

120 

121 # If MSB < MSW, the point estimate for sigma_b is 0 (prevents negative ICC). 

122 var_between = max(0.0, (msb - msw) / n0) 

123 var_within = msw 

124 total_var = var_between + var_within 

125 if total_var == 0: 

126 return 0.0 

127 

128 return float(np.clip(var_between / total_var, 0.0, 1.0))

131def calculate_icc_from_dataframe( 

132 df: pd.DataFrame, 

133 *, 

134 cluster_column: str, 

135 outcome_column: str, 

136) -> float: 

137 """ 

138 Calculate intraclass correlation (ICC) from a dataframe. 

139 

140 Uses a one-way random-effects ANOVA estimator, equivalent to the variance-component 

141 ICC from a linear mixed model with a random intercept per cluster and no fixed 

142 effects beyond the grand mean. 

143 

144 Args: 

145 df: DataFrame containing cluster and outcome columns 

146 cluster_column: Name of the column with cluster identifiers 

147 outcome_column: Name of the column with numeric outcomes 

148 

149 Returns: 

150 ICC value between 0 and 1 

151 

152 Raises: 

153 ValueError: If dataframe is empty or has insufficient data 

154 """ 

155 if len(df) == 0: 

156 raise ValueError("Cannot calculate ICC from empty dataframe") 

157 

158 missing = [name for name in (cluster_column, outcome_column) if name not in df.columns] 

159 if missing: 

160 raise ValueError(f"DataFrame is missing columns: {', '.join(missing)}") 

161 if df[cluster_column].nunique() < 2: 

162 raise ValueError("Need at least 2 clusters to calculate ICC") 

163 if df[cluster_column].isna().any(): 

164 raise ValueError(f"cluster column '{cluster_column}' contains NaN values") 

165 if df[outcome_column].isna().any(): 

166 raise ValueError(f"outcome column '{outcome_column}' contains NaN values") 

167 

168 # Note: If we need to add covariates, we can use a mixed model ICC here if the dataset is large enough. 

169 return _icc_one_way_random_intercept(df, cluster_column=cluster_column, outcome_column=outcome_column)
