26# A warehouse that is briefly unavailable is retried within minutes. One that stays broken, such as with revoked 

27# credentials or a dropped table, settles at hourly pulls, since each failing pull holds the experiment's draw locks 

28# and a worker slot for up to the pull timeout, and leaves a thread stuck on the warehouse call. 

29DWH_PULL_BACKOFF = Backoff( 

30 delays=( 

31 timedelta(minutes=1), 

32 timedelta(minutes=2), 

33 timedelta(minutes=5), 

34 timedelta(minutes=15), 

35 timedelta(minutes=30), 

36 timedelta(hours=1), 

37 ) 

38) 

39# Autofail failures are quick database errors or bugs that hold nothing external, so they keep retrying every 2 

40# minutes. 

41AUTOFAIL_BACKOFF = Backoff(delays=(timedelta(minutes=1), timedelta(minutes=2)))

122 ) 

123 # Graceful shutdown: SIGINT and SIGTERM only set this event, so the worker stops starting new jobs 

124 # and drains active ones instead of dying mid-job. 

125 stop = Event() 

126 previous = {sig: signal.signal(sig, lambda *_: stop.set()) for sig in (signal.SIGINT, signal.SIGTERM)} 

127 exit_code = 1 

128 try: 

129 clock = SystemClock() 

130 pacer = Pacer(poll_interval=poll_interval, clock=clock, exit_after_jobs=exit_after_jobs) 

131 # Each worker will get its own connection, plus one for scheduling. Leaving the executor's context drains 

132 # the running attempts, before the database is torn down. 

133 with database.setup(pool_size=parallelism + 1), AttemptExecutor(parallelism, clock) as executor: 

134 run_worker(jobs, stop, pacer, executor) 

135 exit_code = 0 

136 except BaseException as exc: 

137 # exit_now discards the exception, so report it here. 

138 logger.opt(exception=exc).error("Worker stopped unexpectedly.") 

139 sentry_sdk.capture_exception(exc) 

140 finally: 

141 for sig, handler in previous.items(): 

142 signal.signal(sig, handler) 

143 exit_now(exit_code)

146def exit_now(exit_code: int) -> None: 

147 """Exit at once, killing threads left behind by abandoned warehouse calls. 

148 

149 A warehouse call that outlives its timeout keeps running on a helper thread, possibly forever. A normal 

150 interpreter exit joins every executor thread and would wait for them, so flush what we can and exit without 

151 running exit handlers, even when an exception is propagating. Railway's ALWAYS restart policy 

152 (railway.worker.json) then starts a fresh process. 

153 """ 

154 logger.info(f"Worker exiting with status {exit_code}.") 

155 sentry_sdk.flush() 

156 sys.stdout.flush() 

157 sys.stderr.flush() 

158 os._exit(exit_code)
