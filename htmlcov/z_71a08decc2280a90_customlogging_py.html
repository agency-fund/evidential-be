49def _describe_granian_access(request: Mapping[str, typing.Any]) -> tuple[str, dict[str, typing.Any]]: 

50 """Describes a request from Granian's access log as a message and the fields to log alongside it. 

51 

52 Granian calls its access logger with a format string and a dict of request values. We ignore the format string and 

53 describe the request ourselves: structured logs get each value as a top-level field, so Railway can filter on them, 

54 and other formats get a single line of key=value pairs. 

55 

56 Granian doesn't escape the values, and the path is percent-decoded, so it can contain quotes, spaces, newlines, and 

57 terminal escape sequences. Fields are safe because JSON escapes them; text is quoted with _logfmt_value. 

58 """ 

59 path = request["path"] 

60 query = request["query_string"] 

61 duration_ms = round(request["dt_ms"], 3) 

62 if flags.LOG_FORMAT == LogFormat.STRUCTURED_RAILWAY: 

63 fields = { 

64 "http_method": request["method"], 

65 "http_path": path, 

66 "http_query": query, 

67 "http_status": request["status"], 

68 "http_duration_ms": duration_ms, 

69 "http_protocol": request["protocol"], 

70 "client_addr": request["addr"], 

71 } 

72 return f"{request['method']} {_logfmt_value(path)} {request['status']}", fields 

73 pairs = { 

74 "dur": duration_ms, 

75 "ip": request["addr"], 

76 "method": request["method"], 

77 "path": path, 

78 "q": query, 

79 "code": request["status"], 

80 } 

81 return " ".join(f"{key}={_logfmt_value(value)}" for key, value in pairs.items() if value != ""), {}

87def _logfmt_value(value: object) -> str: 

88 """Formats a value for a key=value log line the way logfmt does. 

89 

90 Values made only of printable characters other than space, '"', '=', and backslash are written as is. Others are 

91 quoted, with quotes, backslashes, and non-printable characters (including Unicode format characters such as 

92 bidirectional overrides) backslash-escaped, so that a value can't break the line apart or alter a terminal. 

93 """ 

94 text = str(value) 

95 if text and all(char.isprintable() and char not in ' "=\\' for char in text): 

96 return text 

97 return '"' + "".join(map(_logfmt_escape, text)) + '"'

100def _logfmt_escape(char: str) -> str: 

101 if escape := _LOGFMT_ESCAPES.get(char): 

102 return escape 

103 if char.isprintable(): 

104 return char 

105 code_point = ord(char) 

106 return f"\\u{code_point:04x}" if code_point <= 0xFFFF else f"\\U{code_point:08x}"

125def _record_to_railway_json(record: loguru_Record): 

126 structured: dict[str, typing.Any] = { 

127 # Railway only lets the Log Explorer filter on top-level fields, so contextual values (from logger.bind(), 

128 # logger.contextualize(), and message format arguments) go at the top level. The fields below take precedence 

129 # over any of them with the same name. 

130 **record["extra"], 

131 "timestamp": record["time"].isoformat(), 

132 "message": record["message"], 

133 "level": record["level"].name, 

134 "logger": record["name"], 

135 "module": record["module"], 

136 "function": record["function"], 

137 "line": record["line"], 

138 "process_id": record["process"].id, 

139 "process_name": record["process"].name, 

140 "thread_id": record["thread"].id, 

141 "thread_name": record["thread"].name, 

142 **_RAILWAY_IDENTITY, 

143 } 

144 if (exc := record["exception"]) is not None: 

145 structured.update({ 

146 "exc_type": str(exc.type.__name__) if exc.type else None, 

147 "exc_msg": str(exc.value) if exc.value else None, 

148 "exc_tb": traceback.format_exception(exc.value, limit=15, chain=True) if exc.value else None, 

149 }) 

150 # default=str and OPT_NON_STR_KEYS avoids disappearing log lines if any context variables are of an unusual type. 

151 try: 

152 return orjson.dumps(structured, default=str, option=orjson.OPT_NON_STR_KEYS).decode() 

153 except orjson.JSONEncodeError: 

154 # orjson rejects integers beyond 64 bits and strings that aren't valid UTF-8, and default= doesn't apply to 

155 # either. json.dumps accepts both, so fall back to it rather than lose the line. 

156 return json.dumps(structured, default=str)
