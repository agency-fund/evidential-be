11from xngin.apiserver.routers.common_api_types import Filter 

12from xngin.stats.cluster_icc import calculate_icc_from_dataframe 

13from xngin.stats.stats_errors import StatsPowerError

16def calculate_cluster_stats_from_database( 

17 session: Session, 

18 sa_table: Table, 

19 cluster_column: str, 

20 outcome_columns: Sequence[str], 

21 filters: list[Filter], 

22) -> dict[str, dict[str, float]]: 

23 """ 

24 Calculate ICC and cluster statistics for one or more metrics from a DWH table. 

25 

26 Fetches all metrics' outcome data in a single DWH query, then computes the stats in 

27 pandas. Orchestrates DWH queries with stats calculations, keeping the stats layer 

28 free of SQLAlchemy dependencies. 

29 

30 Args: 

31 session: SQLAlchemy session for the DWH 

32 sa_table: SQLAlchemy Table object 

33 cluster_column: Column name containing cluster IDs 

34 outcome_columns: Column names containing outcome values 

35 filters: List of filters to apply 

36 

37 Returns: 

38 dict keyed by outcome column name, each value a dict with keys: 

39 icc, avg_cluster_size, cv 

40 """ 

41 data = get_cluster_outcome_data(session, sa_table, cluster_column, outcome_columns, filters) 

42 df = pd.DataFrame(data) 

43 

44 # Cluster sizes and CV are computed over every row with a cluster key, including rows 

45 # whose outcome values are null: metrics are outcomes that may be filled in as the 

46 # experiment runs, so the analysis-time cluster size is the full-population size. 

47 # Do not narrow this to each outcome's non-null rows. 

48 sizes = df.groupby(cluster_column).size() 

49 avg_cluster_size = float(sizes.mean()) 

50 # ddof=0 is the population stddev, matching the SQL stddev_pop this replaced. 

51 cv = float(sizes.std(ddof=0) / sizes.mean()) 

52 

53 stats_by_outcome: dict[str, dict[str, float]] = {} 

54 for outcome_column in outcome_columns: 

55 # ICC uses only rows with a value for this outcome, mirroring the per-metric 

56 # "outcome IS NOT NULL" filter used when each metric was queried separately. 

57 outcome_df = df[[cluster_column, outcome_column]].dropna(subset=[outcome_column]) 

58 if outcome_df.empty: 

59 raise LateValidationError( 

60 f"No data found for cluster column '{cluster_column}' and outcome '{outcome_column}'" 

61 ) 

62 try: 

63 icc = calculate_icc_from_dataframe(outcome_df, cluster_column=cluster_column, outcome_column=outcome_column) 

64 except ValueError as verr: 

65 raise StatsPowerError.from_error(verr, outcome_column) from verr 

66 stats_by_outcome[outcome_column] = { 

67 "icc": icc, 

68 "avg_cluster_size": avg_cluster_size, 

69 "cv": cv, 

70 } 

71 

72 return stats_by_outcome
