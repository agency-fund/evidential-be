1"""Checks that every route's path parameters are consumed by the handler or its 

2dependencies. 

3 

4Route path parameters are declared in the path template using {brackets}; they 

5are mapped to Python variables via method arguments. The decorator is within a 

6few lines of the argument's declaration, making this an intuitive and eloquent 

7way to declare the mappings. However, the path parameters can also be declared 

8on a dependency of the handler, whose implementation may be in a different 

9module. That causes readability issues because the declaration and usage are no 

10longer coincident. If the template parameters and the handler arguments don't 

11match, it will also be a source of bugs. 

12 

13This test mitigates the risk of the reduced readability by mechanistically 

14verifying the path template matches the full tree of the handler's dependencies. 

15This detects two errors: 

16 

17- An argument referring to a path parameter that doesn't exist. Requests then 

18 fail with a 422 naming a path parameter that cannot be supplied, and the 

19 OpenAPI schema advertises a parameter that isn't in the path template. 

20- A path parameter not consumed by an argument. E.g. a route under 

21 /organizations/{organization_id} whose handler takes the datasource-scoped 

22 dependency instead of the organization-scoped one drops the organization 

23 constraint without any symptoms, which would probably be a bug. 

24 

25Ruff's FAST003 only does half of that: it detects mismatches between the 

26template and the handler arguments, but does not traverse the handler's 

27arguments to verify the transitive closure of dependencies. 

28 

29Note this is only matching by "name" of arguments, not type -- it is always up 

30to the implementers of the handlers to ensure the parameters match desired 

31functionality. Viewing the generated OpenAPI spec or the generated API docs 

32is a good way to verify that the API makes sense. 

33"""

45# A route that only the recursive walk below can reach, so that a change to how routers are included 

46# fails this test rather than silently emptying it. 

47CANARY_ROUTE = "/v1/m/organizations/{organization_id}/datasources/{datasource_id}/experiments/{experiment_id}/snapshots"

50def _api_routes(routes: Iterable[BaseRoute]) -> Iterator[APIRoute]: 

51 """Yields every APIRoute in the app, descending into included routers.""" 

52 for route in routes: 

53 if isinstance(route, APIRoute): 

54 yield route 

55 elif hasattr(route, "original_router"): 

56 yield from _api_routes(route.original_router.routes) 

57 

58 

59def _mismatches(routes: Iterable[BaseRoute]) -> list[str]: 

60 """Describes every route whose path parameters disagree with what its signatures consume.""" 

61 described = [] 

62 for route in _api_routes(routes): 

63 declared = {p.alias for p in get_flat_params(route.dependant) if isinstance(p.field_info, params.Path)} 

64 _, _, converters = compile_path(route.path) 

65 in_path = set(converters) 

66 if declared == in_path: 

67 continue 

68 detail = [] 

69 if unconsumed := sorted(in_path - declared): 

70 detail.append(f"named in the path but consumed by nothing: {unconsumed}") 

71 if unfillable := sorted(declared - in_path): 

72 detail.append(f"required by the handler or a dependency but absent from the path: {unfillable}") 

73 methods = "/".join(sorted(route.methods or [])) 

74 described.append(f"{route.name} ({methods} {route.path}): {'; '.join(detail)}") 

75 return described

78def test_path_parameters_match_what_handlers_and_dependencies_consume(): 

79 covered = {route.path for route in _api_routes(app.routes)} 

80 assert CANARY_ROUTE in covered, "The route walk missed the admin router; _api_routes needs updating." 

81 

82 mismatches = _mismatches(app.routes) 

83 assert not mismatches, "Routes whose path parameters disagree with their signatures:\n" + "\n".join(mismatches)
