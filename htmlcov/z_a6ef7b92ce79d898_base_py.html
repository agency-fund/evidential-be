13@dataclass(slots=True, frozen=True) 

14class Backoff: 

15 """Delays after the first, second, third, and later consecutive failures.""" 

16 

17 delays: tuple[timedelta, ...] 

18 

19 @classmethod 

20 def constant(cls, delay: timedelta) -> Backoff: 

21 """Wait the same delay after every failure, however many in a row.""" 

22 return cls(delays=(delay,)) 

23 

24 def delay_after(self, failures: int) -> timedelta: 

25 return self.delays[min(max(failures - 1, 0), len(self.delays) - 1)]

73 def next_run_at(self, now: ColumnElement[datetime]) -> ColumnElement[datetime | None]: 

74 """Return a SQL expression for when the job should next run for an experiment, or NULL when it has no work. 

75 

76 The expression may refer to tables.Experiment and to tables.JobStatus, this job's status row for the 

77 experiment. The worker evaluates it only for committed experiments, so it need not check the state. The 

78 status row is outer-joined, so its columns are NULL until the job first runs. It may read other tables, such 

79 as draws, through correlated subqueries. now is the time the expression is evaluated at. 

80 

81 Nothing about the result is stored. The worker evaluates the expression on every look for work, which happens 

82 at least once per polling interval and again after each attempt finishes, and once more under the 

83 experiment's claim before running the job. A time at or before now means the job is due. The worker delays the 

84 run to the status row's retry_not_before when that is later, so the expression need not consider backoff. 

85 

86 NULL does not stop the job for good: it only means the job doesn't run while the expression stays NULL. As 

87 soon as the rows it reads change so that it gives a time, such as when an experiment is committed or a draw 

88 is inserted, the next look schedules the job. 

89 """

93 """Do one unit of the job's work for the experiment, in the caller's transaction. 

94 

95 The experiment is the row the caller claimed. 

96 

97 To fail, raise any exception, or return JobFailed. A raised exception rolls back the job's writes, is

100 way the failure counts toward the job's backoff.

107def next_run_in_window( 

108 now: ColumnElement[datetime], 

109 interval: timedelta, 

110 *, 

111 opens: SQLColumnExpression[datetime], 

112 closes: SQLColumnExpression[datetime], 

113) -> ColumnElement[datetime | None]: 

114 """Return SQL for when a job that runs every interval within a window should next run, or NULL if it shouldn't. 

115 

116 The job's last success and last attempt are read from its JobStatus row. A job that has never succeeded is due 

117 when the window opens, or now if it is already open, and never once it has closed. After a success, the next 

118 run is due interval later, but no earlier than opens and no later than closes, so the window always ends with a 

119 run. A success at or after closes ends the runs. 

120 

121 Once the window has closed, a run that came due inside it still gets one attempt, however late, if none has 

122 started since it came due. There are no retries after the close. 

123 """ 

124 last_success_at = tables.JobStatus.last_success_at 

125 last_run_at = tables.JobStatus.last_run_at 

126 # Name the window's bounds and the next run after the last success once per row, so the CASE below reads them 

127 # instead of repeating their arithmetic. The next run is kept inside the window: one that would come before the 

128 # window opens waits for it to open, and one that would overshoot the close runs at the close, so the window 

129 # still gets a final run. 

130 window = values( 

131 column("opens", DateTime(timezone=True)), 

132 column("closes", DateTime(timezone=True)), 

133 column("after_success", DateTime(timezone=True)), 

134 name="run_window", 

135 ).data([(opens, closes, func.least(func.greatest(last_success_at + interval, opens), closes))]) 

136 return ( 

137 select( 

138 case( 

139 # The window closed before the job ever succeeded. 

140 (and_(last_success_at.is_(None), now > window.c.closes), None), 

141 # The job has never succeeded, so it's due now, or when the window opens if that's later. 

142 (last_success_at.is_(None), func.greatest(now, window.c.opens)), 

143 # The job already ran at the close, which is the window's final run. 

144 (last_success_at >= window.c.closes, None), 

145 # The window has closed, and an attempt started after the run came due without succeeding, since a 

146 # success would have moved last_success_at past it. Either the final run at the close was attempted 

147 # after the close and failed, or a run due inside the window failed and its backoff retry would come 

148 # after the close. Failed runs aren't retried once the window has closed. 

149 (and_(now > window.c.closes, last_run_at >= window.c.after_success), None), 

150 else_=window.c.after_success, 

151 ).label("run_at") 

152 ) 

153 .select_from(window) 

154 .scalar_subquery() 

155 )
