1"""Configuration of the OIDC relying party, parsed from environment variables. 

2 

3One identity provider (IDP) is configured at a time. Any provider implementing OpenID Connect Discovery and the 

4authorization code flow with PKCE can be used; provider differences are handled by spec-conformant defaults rather 

5than provider-specific switches. 

6"""

14# Principal fields that XNGIN_OIDC_CLAIM_MAP may populate from ID token claims. 

15ALLOWED_CLAIM_MAP_TARGETS = frozenset({"hd"}) 

16# The scope request parameter. openid requests an ID token, and email adds the email claim used to look up users. 

17SCOPE = "openid email" 

18DISCOVERY_PATH = "/.well-known/openid-configuration"

22 pass

25def parse_claim_map(raw: str) -> dict[str, str]: 

26 """Parses "claim:field,claim:field" into a mapping of ID token claim name to Principal field name.""" 

27 claim_map: dict[str, str] = {} 

28 for entry in (item.strip() for item in raw.split(",")): 

29 if not entry: 

30 continue 

31 claim, separator, target = (part.strip() for part in entry.partition(":")) 

32 if not separator or not claim or not target: 

33 raise OidcMisconfiguredError( 

34 f"{flags.ENV_XNGIN_OIDC_CLAIM_MAP} entry '{entry}' is not in claim:field form." 

35 ) 

36 if target not in ALLOWED_CLAIM_MAP_TARGETS: 

37 raise OidcMisconfiguredError( 

38 f"{flags.ENV_XNGIN_OIDC_CLAIM_MAP} target '{target}' is not a Principal auxiliary field. " 

39 f"Allowed: {', '.join(sorted(ALLOWED_CLAIM_MAP_TARGETS))}." 

40 ) 

41 if target in claim_map.values(): 

42 raise OidcMisconfiguredError(f"{flags.ENV_XNGIN_OIDC_CLAIM_MAP} maps more than one claim to '{target}'.") 

43 claim_map[claim] = target 

44 return claim_map

47def check_absolute_https_url(value: str, *, allow_http: bool = False, allow_query: bool = True) -> str | None: 

48 """Explains why value is not an absolute https URL that is safe to use, or returns None when it is. 

49 

50 Credentials and fragments are never allowed: OAuth endpoint and redirect URIs must not carry a fragment 

51 (RFC 6749 sections 3.1 and 3.1.2), and a URL with credentials is never something to hand to a browser. 

52 """ 

53 if any(character.isspace() or ord(character) < 0x20 or character == "\x7f" for character in value): 

54 return "must not contain whitespace or control characters" 

55 parsed = urlsplit(value) 

56 if parsed.scheme not in (("https", "http") if allow_http else ("https",)): 

57 return "must be an https:// URL" 

58 try: 

59 _ = parsed.port 

60 except ValueError: 

61 return "has an invalid port" 

62 if not parsed.hostname: 

63 return "must include a hostname" 

64 if parsed.username is not None or parsed.password is not None: 

65 return "must not include credentials" 

66 if parsed.fragment or value.endswith("#"): 

67 return "must not include a fragment" 

68 if not allow_query and (parsed.query or value.endswith("?")): 

69 return "must not include a query" 

70 return None

73def _require(value: str, env_var: str) -> str: 

74 value = value.strip() 

75 if not value: 

76 raise OidcMisconfiguredError(f"{env_var} environment variable is not set.") 

77 return value

80@dataclass(frozen=True, slots=True) 

81class OidcSettings: 

82 """Describes the identity provider this server trusts.""" 

83 

84 issuer: str 

85 client_id: str 

86 redirect_uri: str 

87 # Maps ID token claim names to Principal auxiliary field names (e.g. {"hd": "hd"}). 

88 claim_map: Mapping[str, str] = field(default_factory=dict) 

89 # Unset for public clients, which authenticate with PKCE alone. 

90 client_secret: str | None = None

92 def discovery_url(self) -> str: 

93 # OpenID Connect Discovery 1.0 section 4.1: a terminating slash on the issuer is removed before appending the 

94 # well-known path. The issuer itself is compared exactly as configured (authentik's issuers end with a slash). 

95 return self.issuer.rstrip("/") + DISCOVERY_PATH 

96 

97 @classmethod 

98 def parse( 

99 cls, 

100 *, 

101 issuer: str, 

102 client_id: str, 

103 client_secret: str, 

104 redirect_uri: str, 

105 claim_map: str, 

106 allow_http: bool, 

107 ) -> OidcSettings: 

108 """Builds settings from raw environment variable values, raising OidcMisconfiguredError on invalid input.""" 

109 issuer = _require(issuer, flags.ENV_XNGIN_OIDC_ISSUER) 

110 # OpenID Connect Discovery 1.0 section 3: the issuer is an https URL with no query or fragment. 

111 if reason := check_absolute_https_url(issuer, allow_http=allow_http, allow_query=False): 

112 raise OidcMisconfiguredError(f"{flags.ENV_XNGIN_OIDC_ISSUER} {reason} (got {issuer!r}).") 

113 redirect_uri = _require(redirect_uri, flags.ENV_XNGIN_OIDC_REDIRECT_URI) 

114 if reason := check_absolute_https_url(redirect_uri, allow_http=allow_http): 

115 raise OidcMisconfiguredError(f"{flags.ENV_XNGIN_OIDC_REDIRECT_URI} {reason} (got {redirect_uri!r}).") 

116 return cls( 

117 issuer=issuer, 

118 client_id=_require(client_id, flags.ENV_XNGIN_OIDC_CLIENT_ID), 

119 redirect_uri=redirect_uri, 

120 claim_map=parse_claim_map(claim_map), 

121 client_secret=client_secret.strip() or None, 

122 ) 

123 

124 @classmethod 

125 def from_flags(cls) -> OidcSettings: 

126 return cls.parse( 

127 issuer=flags.OIDC_ISSUER, 

128 client_id=flags.OIDC_CLIENT_ID, 

129 client_secret=flags.OIDC_CLIENT_SECRET, 

130 redirect_uri=flags.OIDC_REDIRECT_URI, 

131 claim_map=flags.OIDC_CLAIM_MAP, 

132 allow_http=flags.is_dev_environment(), 

133 )
